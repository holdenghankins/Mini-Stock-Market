stock:public security {
    float revenue
    float expenditures
    float assets
    array<float, 2> growth
    float hype
    int numLeft

    void simDay {
        currentDay++
        applyGrowth()
        calcPrice()
        trade()
        calcHype()
    }

    void applyGrowth() {
        let g be a range of values growth[0] -- growth[1]
        generate 3 random numbers, g1 g2 g3, ∈ g
        add g1 to revenue
        add -g2 to expenditures
        add g3/4 to assets, subtract from revenue
    }

    void calcPrice() {
        return = revenue * hype / numLeft
    }

    void trade() {
        numLeft = (revenue * hype) / price
    }

    float calcHype() {
        let change = (getChange(1) + getChange(5) + getChange(20))
        hype += change
    }
}
